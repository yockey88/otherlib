cmake_minimum_required(VERSION 3.27)

project(${project-name} VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CONFIGURATION_TYPES "Debug;Release;Profile;ProfileD" CACHE STRING "Build types" FORCE)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

## must match the SDK's dynamic CRT (the installed otherlib libraries are built /MD[d])
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<OR:$<CONFIG:Debug>,$<CONFIG:ProfileD>>:Debug>DLL")

## provides otherlib targets plus the add_static_driver/add_dynamic_driver macros,
##  which set OTHER_CLIENT, the platform defines, and the otherlib link
find_package(otherlib REQUIRED)

###
##  A dynamic driver module (loaded by a host at runtime)
##
add_dynamic_driver(${project-name}
  src/${project-name}.cpp
  src/${project-name}_driver.cpp
)
target_compile_options(${project-name} PRIVATE /W4 /permissive-)
