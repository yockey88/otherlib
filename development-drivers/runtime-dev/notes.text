enum EnvCalls {
  SET_SCENE = 0 ///< takes 1 argument (an address) and reads a 64 bit unsigned id and sets it as the active scene, unloading any currently playing scene if necessary
}

set_scene:
  write N, <id>
  load 00, EnvCalls::SET_SCENE
  load 01, N
  envcall
